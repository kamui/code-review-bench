1236: 	if acCtx.Err() != nil {
1237: 		ac.mu.Unlock()
1238: 		return
1239: 	}

1301: 	ac.mu.Lock()
1302: 	ac.backoffIdx = 0
1303: 	ac.mu.Unlock()
1304: }
